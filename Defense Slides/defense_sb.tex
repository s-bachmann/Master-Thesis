\titleframe

\begin{frame}{Overview}
    \begin{itemize}
        \item Created 3 different models and 2 scenarios for crack propagation in snow using SALVUS
        \item Two pure spectral element (SE) models, one combined with data from material point method (MPM) simulations
        \item \textbf{Goal}: Couple dynamic snow modelling with SE to model crack propagation in snow
    \end{itemize}
    
\end{frame}

\begin{frame}{Nomenclature \& Abbreviations}
    \begin{itemize}
        \item \textbf{SE, FE, FEM}: Spectral elements, Finite Element (Method), done in SALVUS
        \item \textbf{MPM}: Material Point Method Numerical Modeling
        \item \textbf{PST}: Propagation Saw Test
        \item \textbf{F-k}: Frequency-Wavenumber Plot
        \item \textbf{DAS}: Distributed Acoustic Sensing (Fibre-Optics)
        \item \textbf{STF}: Source-time function (the wavelet fired by SALVUS)
    \end{itemize}
    
\end{frame}

\begin{frame}{Why model crack propagation in snow? - Relevance}
    \begin{itemize}
        \item 22 fatalities by avalanche in Switzerland every year \parencite{LongtermStatistics}
        \item Avalanches are danger for human life but also infrastructure \parencite{schweizerSnowAvalancheFormation2003}
        \item Understanding processes associated with avalanches important
        \item Avalanche release because of loading, leading to crack propagation in a weal layer \parencite{gaumeSnowFractureRelation2017a}
    \end{itemize}
    
\end{frame}

\begin{frame}{Relevance}
\begin{itemize}
    \item Crack propagation at different speeds: sub-Rayleigh (below vs of material) and Supershear (above vs) \parencite{trottetTransitionSubRayleighAnticrack2022}, transition between two regimes possible \parencite{bergfeldSignaturesSubRayleighSupershear2025}
        \item Fast crack propagation \rightarrow critical crack length for release \parencite{gaumeModelingCrackPropagation2015} is reached faster
        \item Distributed Acoustic Sensing (DAS) already used to monitor avalanches in Swizterland \parencite{paitzPhenomenologyAvalancheRecordings2023,edmeFiberopticDetectionSnow2023} and Norway \parencite{turquetAutomatedSnowAvalanche2024}
        \item No knowledge if crack propagation associated with breakoff can be seen in DAS data
\end{itemize}
\end{frame}

\begin{frame}{Research Questions}
    \begin{itemize}
        \item What FE framework is required to to simulate DAS measurements of crack propagation in snow, and what are the key assumptions and limitations?
        \item What physical phenomena are observed in a two-layer model with a simple STF and how do they differ for sub-Rayleigh versus Supershear crack propagation?
        \item To what extent do the results change if source wavelets from a MPM simulation of a propagation saw test (PST) are injected?
        \item How could the phenomena observed be used to distinguish crack propagation regimes in real-world DAS data and aid in detecting snow slab avalanche breakoff?
    \end{itemize}
    
\end{frame}

\begin{frame}{Results}
    \begin{itemize}
        \item SALVUS Output: Displacement field u
        \item $\ \frac{\partial u_i}{\partial t}$ \rightarrow velocity in x and y
        \item $\ \frac{\partial u_i}{\partial j}$ \rightarrow strain in x and y $\ \epsilon$
        \item $\ \frac{\partial^2 u_i}{\partial t \partial t}$ \rightarrow strain rate $\ \dot{\epsilon}$
        \item Simulation od DAS data from $\ \epsilon$, $\ \dot{\epsilon}$ using Pyber Gauge length class by \cite{noeFrontiersFullwaveformSeismology2025}, gauge length acoording to \cite{turquetAutomatedSnowAvalanche2024}
        \item Spate-time plots of strain, strain rate and velocity at different reciever depths
        \item Frequency-wavenumber plots with theoretical speeds and dispersion curves
        \item Animations of velocity wavefields
    \end{itemize}
    
\end{frame}

\begin{frame}[allowframebreaks]{Simple Sub-Rayleigh: Strain and Strain Rate}
    \begin{figure}[H]
        \centering
        \includegraphics[width=0.7\linewidth]{../thesis_salome_bachmann_from_particles_to_waves/figures/preliminary_sub_strain.png}
        \caption{Horizontal Strain and strain rate for sub-Rayleigh crack propagation in simple two-layer model. Plotted at crack depth (2.625m) and at the snow bottom (2.25m). Gaugle length is 10.4m. }

    \end{figure}
    \begin{itemize}
        \item Two-wavefront structure
        \item Main (thicker) wavefront: slope corresponds to prescribed rupture speed
        \item Second wavefront: coming only from first source
        \item Wavefront is emitted by all sources but they interfer \parencite{madariagaDynamicsExpandingCircular1976}, as shown by wider source spacing
        \item Slope of line close to p-wave speed in medium: S0 wave bounded by plate wave speed limit \parencite{roseWavesPlates2014}
        \item S0/P-wave precursor observed in field data by \cite{herwijnenSeismicSensorArray2011}
        \item Oscillatory behavior of both wavefronts (polarity flip at adjacent timesteps)
        \item At rupture arrest (270m): v-shaped arrest signature: bidirectional, at fastest speed in medium \parencite{freundDynamicFractureMechanics1998}
    \end{itemize}
    
\end{frame}

\begin{frame}[allowframebreaks]{Simple Supershear: Strain and Strain Rate}
    \begin{figure}[H]
        \centering
        \includegraphics[width=0.7\linewidth]{../thesis_salome_bachmann_from_particles_to_waves/figures/preliminary_super_strain.png}
        \caption{Horizontal Strain and strain rate for Supershear crack propagation in simple two-layer model. Plotted at crack depth (2.625m) and at the snow bottom (2.25m). Gaugle length is 10.4m. }
    \end{figure}
    \begin{itemize}
        \item One wavefront, contrary to sub-Rayleigh
        \item Slope corresponts to rupture speed (within 1m/s of vp and S0)
        \item Dispersion and broadening of main wavefront \parencite{kAkiRichards19801981}
        \item Oscillatory wavefront
        \item At x=270m, unidirectional fan-like arrest front: rupture outruns vs, so arrest energy is only propagated forward \parencite{madariagaHighfrequencyRadiationCrack1977}
        \item S0 precursor can be seen in traces
    \end{itemize}
    
\end{frame}

\begin{frame}[allowframebreaks]{MPM-Guided: Strain and Strain Rate}
    \begin{figure}[H]
        \centering
        \includegraphics[width=0.7\linewidth]{../thesis_salome_bachmann_from_particles_to_waves/figures/mpm_strain.png}
        \caption{Horizontal Strain and strain rate for MPM-guided model. Plotted at crack depth (0.525) and at the snow bottom (0.3m). Gaugle length is 10.4m. }
    \end{figure}
    \begin{itemize}
        \item One two main wavefronts observed
        \item Crack extent from 75m to 125m: sub-Rayleigh speed from front
        \item Positive wedge in crack area: extends until rupture arrest, then jumps to same thickness as negative front \rightarrow quasi-static offset
        \item Backward-propagating front and forward after crack arrest: close to vp/S0 from slope
        \item Second, weaker wavefront: at onset but also arrest: same arrest structure as sub-Rayleigh
        \item Oscillatory wavefronts
    \end{itemize}
    
\end{frame}

\begin{frame}{Quasi-Static Offset}
    \begin{itemize}
        \item Static field represents permanent deformation after sources have fired \parencite{zhuNoteDynamicStatic2002,archambeauGeneralTheoryElastodynamic1968}
        \item MPM STFS are not symmetric: contrary to Ricker wavelets, they do not integrate to 0 over time \rightarrow net displacement of material \parencite{zhuNoteDynamicStatic2002,kAkiRichards19801981}
        \item Cumulative integral of MPM STFs is negative: collapse of the weak layer, as observed by \parencite{heierliAnticracksNewTheory2003, heierliAnticrackNucleationTriggering2008,gaumeDynamicAnticrackPropagation2018,trottetTransitionSubRayleighAnticrack2022}
        \item All components of MPM STF (directionality) indicate mixed-mode anticrack instead of pure opening or shearing (mode I, II or II)
        
    \end{itemize}
    
\end{frame}

\begin{frame}[allowframebreaks]{Simple model: Lamb Wave Identification}
    \begin{itemize}
        \item Polarity oscillation in all plots: indicative of Lamb waves
        \item Plotting dispersion curves over f-k spectra: energy is concentrated at symmetric (S0) and anitsymetric (A0) mode
        \item Antisymmetric mode gets excited preferentially when source is applied normal to midplane of material \parencite{lambWavesElasticPlate1917,roseWavesPlates2014,achenbachWavePropagationElastic2012}, so in this case yy
