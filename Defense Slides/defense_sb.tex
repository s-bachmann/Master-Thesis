\begin{frame}{Why Model Crack Propagation in Snow? - Relevance}
    \begin{itemize}
        \item 22 fatalities by avalanche in Switzerland every year \parencite{LongtermStatistics}
        \item Avalanches are danger for human life but also infrastructure \parencite{schweizerSnowAvalancheFormation2003}
        \item Understanding processes associated with avalanches important
    \end{itemize}
    
\end{frame}

\begin{frame}{Relevance}
\begin{itemize}
    \item Avalanche release because of loading, leading to crack propagation in a weak layer \parencite{gaumeSnowFractureRelation2017a}
    \item Crack propagation at different speeds: sub-Rayleigh (below vs of material) and Supershear (above vs) \parencite{trottetTransitionSubRayleighAnticrack2022}, transition between two regimes possible \parencite{bergfeldSignaturesSubRayleighSupershear2025}
        \item Fast crack propagation $\rightarrow$ critical crack length for release \parencite{gaumeModelingCrackPropagation2015} is reached faster \par
        \item Distributed Acoustic Sensing (DAS) already used to monitor avalanches in Switzerland \parencite{paitzPhenomenologyAvalancheRecordings2023,edmeFiberopticDetectionSnow2023} and Norway \parencite{turquetAutomatedSnowAvalanche2024}
        \item No knowledge if crack propagation associated with breakoff can be seen in DAS data
\end{itemize}
\end{frame}

\begin{frame}{Overview}
    \begin{itemize} 
        \item \textbf{Goal}: Couple dynamic snow modelling with SE to model crack propagation in snow
        \item Created 3 different models and 2 scenarios for crack propagation in snow using SALVUS
        \item Two pure spectral element (SE) models, one combined with data from material point method (MPM) simulations
       
    \end{itemize}
    
\end{frame}

\begin{frame}{Nomenclature \& Abbreviations}
    \begin{itemize}
        \item \textbf{SE, FE, FEM}: Spectral elements, Finite Element (Method), done in SALVUS
        \item \textbf{MPM}: Material Point Method Numerical Modeling % EXPLAIN
        \item \textbf{PST}: Propagation Saw Test
        \item \textbf{F-k}: Frequency-Wavenumber Plot
        \item \textbf{DAS}: Distributed Acoustic Sensing (Fibre-Optics) % Explain what it measures
        \item \textbf{STF}: Source-time function (the wavelet fired by SALVUS)
    \end{itemize}
    
\end{frame}



\begin{frame}{Research Questions}
    \begin{itemize}
        \item What FE modelling framework is required to simulate DAS measurements of crack propagation in snow, and what are the key assumptions and limitations?
        \item What physical phenomena are observed in a two-layer model with a simple STF and how do they differ for sub-Rayleigh versus Supershear crack propagation?
        \item To what extent do the results change if source wavelets from a MPM simulation of a PST are injected?
        \item How could the phenomena observed be used to distinguish crack propagation regimes in real-world DAS data and aid in detecting snow slab avalanche breakoff?
    \end{itemize}
    
\end{frame}

\begin{frame}{Anticrack in MPM}
    \begin{itemize}
        \item Static field represents permanent deformation after sources have fired \parencite{zhuNoteDynamicStatic2002,archambeauGeneralTheoryElastodynamic1968}
        \item MPM STFS are not symmetric: contrary to Ricker wavelets, they do not integrate to 0 over time $\rightarrow$ net displacement of material \parencite{zhuNoteDynamicStatic2002,kAkiRichards19801981}
        \item Cumulative integral of MPM STFs is negative: collapse of the weak layer, as observed by \parencite{heierliAnticracksNewTheory2003, heierliAnticrackNucleationTriggering2008,gaumeDynamicAnticrackPropagation2018,trottetTransitionSubRayleighAnticrack2022}
        \item All components of MPM STF (directionality) indicate mixed-mode anticrack instead of pure opening or shearing (mode I, II or II)
        
    \end{itemize}
    
\end{frame}

\begin{frame}[allowframebreaks]{Summary of Results \& Discussion}
    \begin{itemize}
        \item S0 Precursor observed in all Strain \& Strain Rate Plots, as well as in traces
        \item Main wavfront in all cases is A0 $\rightarrow$ source normal to slab midplane 
        \item Different rupture arrest structures depending on propagation speed
        \item Quasi-static wavefield for sources that do not integrate to 0 over time
        \item Weakening of amplitude only in simple case $\rightarrow$ also thicker slab 
    \end{itemize}
    
\end{frame}

\begin{frame}{Answer to Research Questions - Conclusions}
     \begin{itemize}
        \item \textbf{What physical phenomena are observed in a two-layer model with a simple STF and how do they differ for sub-Rayleigh versus Supershear crack propagation?} Rupture-arrest signatures depending on propagation speed, S0 precursor, A0 main wavefront for both propagation regimes. Changes in amplitude
        \item \textbf{To what extent do the results change if source wavelets from a MPM simulation of a propagation saw test (PST) are injected?} Verification of sub-Rayleigh S0 precursor, A0 mode excitation, changes because appearance of quasi-static offset
        \item \textbf{How could the phenomena observed be used to distinguish crack propagation regimes in real-world DAS data and aid in detecting snow slab avalanche breakoff?} Precursors (S0 and quasi-static offset) used as "early" warning, identification of propagation speed using arrest structures, possibility of identifying transitions by number of wavefronts
    \end{itemize}
    
\end{frame}
